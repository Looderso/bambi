# The golden scenarios, read by BOTH tools/golden.sh (hash them, within one toolchain) and
# tools/check-portable.sh (render them twice, on two architectures, and null them). One list, so a
# scenario added for one check is exercised by the other.
#
#   name | arguments to bambi-render | the trace column that MUST vary
#
# Each scenario exercises a join that unit tests cannot reach: the whole chain from audio through
# features, matrix, clock and trajectory to ambisonics.
#
# The third field is not decoration. A scenario whose modulated destination never actually moves is
# a golden file that locks in a hash while testing nothing, and the first version of this set had
# exactly that: `tonal-extent` ran on pink noise, which reads Tonal ~0, so the extent path was never
# touched. Every scenario declares what it claims to move, and the check fails if it does not move.
static|--noise 2 --order 1 --speed 0 --generator orbit|
orbit-const|--noise 3 --order 3 --speed 90 --width 20 --generator orbit|s
premise-level-speed|--noise 3 --order 3 --speed 0 --generator orbit --mod level:motion.speed=0.5|speed
attack-width|--noise 3 --order 3 --speed 45 --generator lissajous --mod attack:render.width=0.6|width
tonal-extent|--tone 3 --order 3 --speed 60 --generator orbit --mod tonal:transform.extent=-0.5|extent
lfo-synced-yaw|--noise 4 --order 3 --speed 30 --generator orbit --bpm 128 --mod lfo1:transform.yaw=0.5|yaw
env-displace|--noise 3 --order 3 --speed 0 --generator orbit --mod env1:motion.displace=0.4|displace
tone-roll-pitch|--tone 3 --order 3 --speed 40 --generator lissajous --mod level:transform.roll=0.5 --mod tonal:transform.pitch=0.4|roll
gain-duck|--noise 3 --order 1 --speed 0 --generator orbit --mod level:output.gain=-0.3|gain
gain-duck-slow|--noise 3 --order 1 --speed 0 --generator orbit --mod level:output.gain=-0.3 --param level.release=1500|gain
# the input trim reaches the detectors as well as the encoder, as the plugin applies it
trim-level-speed|--noise 3 --order 3 --speed 0 --generator orbit --param input.trim=-18 --mod level:motion.speed=0.5|speed
open-pingpong|--noise 3 --order 3 --speed 120 --generator spiral --mode pingpong|s
order7-wide|--noise 2 --order 7 --speed 75 --width 60 --generator spiral|s
# The paths and regions the scenarios above don't reach. A pentagon is five great-circle edges taken at one speed
# through its corners; a triangle whose radius an LFO sweeps, standing on its first corner, moves only by the
# radius; the wave climbs and falls as it goes round. Catches, for each: a change to how that path or region is
# built or played.
polygon-pentagon|--noise 3 --order 3 --speed 90 --generator polygon --set sides=5 --set radius=50|az
polygon-radius|--noise 3 --order 3 --speed 0 --generator polygon --set sides=3 --mod lfo1:polygon.radius=0.5|el
wave-path|--noise 3 --order 3 --speed 60 --generator wave|el
region-speed|--noise 3 --order 3 --speed 40 --generator orbit --region spot --param region1.size=70 --param region1.softness=40 --param region1.yaw_rate=30 --mod region1:motion.speed=0.8|region
region-band|--noise 3 --order 3 --speed 40 --generator orbit --region band --param region1.thickness=40 --param region1.pitch_rate=30 --mod region1:motion.speed=0.8|region
region-sectors|--noise 3 --order 3 --speed 40 --generator orbit --region sectors --param region1.yaw_rate=45 --mod region1:render.width=0.6|region
region-dots|--noise 3 --order 3 --speed 40 --generator orbit --region dots --param region1.dot_size=25 --param region1.yaw_rate=30 --mod region1:motion.speed=0.8|region
# A stereo input: `--stereo` gives the built-in noise a side, a second noise unrelated to the
# mid. What these reach that no unit test does is two encoders ramping on one control grid -- the
# block-size independence the harness renders five ways -- and the side being fed at all.
input-midside|--noise 3 --order 3 --speed 60 --width 15 --generator orbit --stereo --param input.mode=1 --param input.spread=90|s
input-stereo|--noise 3 --order 3 --speed 60 --width 15 --generator lissajous --stereo --param input.mode=2 --param input.offset=0.2|s
input-spread-moves|--noise 3 --order 3 --speed 30 --generator orbit --stereo --param input.mode=1 --param input.spread=40 --mod level:input.spread=0.6|s
#
# Echo's, rendered by bambi-echo-render (a name beginning echo-). The input is a few voices of noise
# bursts round the listener, or one impulse; `energy` is the output's energy a control step, and p1..p4
# the four periods in samples.
# `--set dry=-60` on every effect scenario, so they hash the wet path alone; reverb-distance is the
# one that renders a dry path, and sets its own.
echo-ping-pong|--order 3 --seconds 3 --set dry=-60 --impulse 35,0 --set tap1.on=0 --set tap4.on=1 --set tap4.feedback=-2|energy
echo-two-taps-swing|--order 3 --seconds 4 --set dry=-60 --set tap2.on=1 --set tap2.swing=0.4|energy
echo-tilted-skew|--order 3 --seconds 4 --set dry=-60 --set tap1.axis_az=40 --set tap1.axis_el=20 --set tap1.skew=0.5 --set tap1.feedback=-1|energy
echo-blur-band|--order 3 --seconds 4 --set dry=-60 --set tap1.blur=30 --set tap1.low_cut=300 --set tap1.high_cut=3000 --set tap1.feedback=-1|energy
echo-sleep-wake|--order 3 --seconds 4 --set dry=-60 --at 1.0 tap1.on=0 --at 2.5 tap1.on=1|energy
echo-tempo-change|--order 3 --seconds 4 --set dry=-60 --set tap1.feedback=-1 --at 1.5 bpm=97|p1
echo-send-front|--order 3 --seconds 4 --set dry=-60 --set send.kind=spot --set send.size=90 --set send.softness=40 --set tap1.feedback=-2|energy
echo-send-band|--order 3 --seconds 4 --set dry=-60 --set send.kind=band --set send.thickness=30 --set tap1.feedback=-2|energy
echo-locate|--order 3 --seconds 4 --set dry=-60 --set tap1.feedback=-0.5 --at 2.0 reset|energy
echo-order7|--order 7 --seconds 2 --set dry=-60 --set tap3.on=1 --set tap3.skew=-0.4|energy

# room works out to. Angles in the arguments are degrees.
reverb-hall-impulse|--order 3 --seconds 3 --set dry=-60 --impulse 0,0|energy

reverb-cathedral|--order 3 --seconds 4 --set dry=-60 --impulse 20,10 --set room.size=38 --set room.shape=tall --set room.decay=6.5 --set room.tone=0.35 --set distance=14|energy
reverb-distance|--order 3 --seconds 3 --set dry=0 --at 1.0 distance=14|dry_gain
reverb-send-front|--order 3 --seconds 3 --set dry=-60 --set send.kind=spot --set send.size=50 --set send.softness=15|energy
reverb-return-band|--order 3 --seconds 3 --set dry=-60 --set return.kind=band --set return.thickness=40|energy
reverb-send-dots|--order 3 --seconds 3 --set dry=-60 --set send.kind=dots --set send.dots=6 --set send.dot_size=25|energy

#
# The Decoder's, rendered by bambi-decode-render (a name beginning decode-). The input is a few voices of
# noise round the listener, or one impulse 10 ms in; the output is the whole bus. `energy` is every output's
# energy a control step, `headphones` and `speakers` each output's own, head_* the tracker's head the decode
# undoes, scene_* the scene's turn. Angles in the arguments are degrees, distances metres.
# Catches: a change to the binaural filters' design, their convolution, or which ear is which.
decode-headphones|--order 3 --seconds 2|headphones
# Catches: the head undone the wrong way round, or not at all, on the way to the binaural filters.
decode-head-turning|--order 3 --seconds 2 --at 0.4 head.yaw=60 --at 1.0 head.yaw=-90 --at 1.0 head.pitch=30|head_yaw
# Catches: the scene's rates not integrated, or their turn not taken back to zero at a start or a locate.
decode-scene-turns|--order 3 --seconds 2 --set yaw_rate=90 --set pitch=20 --at 1.2 reset|scene_yaw
# Ten loudspeakers at ten distances, 1.2 to 2.8 m: each is delayed and attenuated toward the farthest.
# Catches: a loudspeaker's distance delay or gain dropped, or put on another output.
decode-speaker-distances|--order 3 --seconds 1 --impulse 40,10 --set headphones=0 --set speakers=1 --set speaker=30,0,2.5 --set speaker=-30,0,2.2 --set speaker=0,0,2.8 --set speaker=110,0,1.6 --set speaker=-110,0,1.4 --set speaker=180,0,1.9 --set speaker=45,40,1.5 --set speaker=-45,40,1.7 --set speaker=135,40,1.2 --set speaker=-135,40,1.3|speakers
# A layout file's filter matrix: first order, N3D, 44.1 kHz, three outputs, played on a third-order field at 48 kHz.
# Catches: its N3D inputs read as SN3D, or its filters not resampled to the rate.
decode-matrix-file|--order 3 --seconds 1 --set headphones=0 --set speakers=1 --set matrix=core/tests/decode/made-matrix.sofa|speakers
# A first-order field on 7.1.4, and heard through that layout on headphones, which decode the loudspeakers at
# seventh order. Catches: the loudspeakers encoded at the field's order rather than the decode's.
decode-narrow-field|--order 1 --seconds 1 --set speakers=1 --set layout=4+7+0 --set through_layout=1 --set headphone_channel=13|energy
